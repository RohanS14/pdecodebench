Heat Equation Simulation Code Example 1
"""
1D Heat Equation: Explicit FTCS (Forward-Time Central-Space) scheme.
Initial condition: Gaussian pulse.
"""
import numpy as np
import matplotlib.pyplot as plt


# Domain and physical parameters
L, alpha = 1.0, 0.01
nx, nt = 101, 2000
dx = L / (nx - 1)
dt = 0.4 * dx**2 / alpha  # CFL condition for FTCS: alpha*dt/dx^2 <= 0.5
r = alpha * dt / dx**2


x = np.linspace(0, L, nx)
u = np.exp(-((x - L / 2) ** 2) / (2 * 0.02 ** 2))  # Gaussian initial condition
u[0], u[-1] = 0.0, 0.0  # Dirichlet boundary conditions


snapshots = {0: u.copy()}
save_steps = [nt // 4, nt // 2, nt - 1]


for n in range(nt):
    u_new = u.copy()
    u_new[1:-1] = u[1:-1] + r * (u[2:] - 2 * u[1:-1] + u[:-2])  # explicit update
    u_new[0], u_new[-1] = 0.0, 0.0
    u = u_new
    if n in save_steps:
        snapshots[n] = u.copy()


plt.figure(figsize=(7, 5))
for step, snap in snapshots.items():
    plt.plot(x, snap, label=f"t = {step * dt:.4f} s")
plt.xlabel("x")
plt.ylabel("u(x,t)")
plt.title("1D Heat Equation - FTCS (Gaussian IC)")
plt.legend()
plt.tight_layout()
plt.savefig("heat_1d_ftcs_gaussian.png", dpi=150)
plt.show()

"""
1D Heat Equation: Crank-Nicolson (unconditionally stable implicit scheme).
Initial condition: top-hat / step function.
"""
import numpy as np
import matplotlib.pyplot as plt
from scipy.linalg import solve_banded


# Domain and physical parameters
L, alpha = 1.0, 0.01
nx, nt = 101, 500
dx = L / (nx - 1)
dt = 0.001  # No CFL restriction needed; Crank-Nicolson is unconditionally stable
r = alpha * dt / dx**2


x = np.linspace(0, L, nx)
u = np.zeros(nx)
u[(x > 0.4) & (x < 0.6)] = 1.0  # top-hat initial condition
u[0], u[-1] = 0.0, 0.0


# Build tridiagonal system (I - r/2 * D2) u_new = (I + r/2 * D2) u_old
n_int = nx - 2
ab = np.zeros((3, n_int))
ab[0, 1:] = -r / 2
ab[1, :] = 1 + r
ab[2, :-1] = -r / 2


snapshots = {0: u.copy()}
save_steps = [nt // 4, nt // 2, nt - 1]


for n in range(nt):
    rhs = (r / 2) * u[:-2] + (1 - r) * u[1:-1] + (r / 2) * u[2:]  # explicit half-step RHS
    u_int = solve_banded((1, 1), ab, rhs)
    u[1:-1] = u_int
    u[0], u[-1] = 0.0, 0.0
    if n in save_steps:
        snapshots[n] = u.copy()


plt.figure(figsize=(7, 5))
for step, snap in snapshots.items():
    plt.plot(x, snap, label=f"t = {step * dt:.3f} s")

plt.title("1D Heat Equation - Crank-Nicolson (Top-Hat IC)")

Heat Equation Simulation Code Example 3
"""
2D Heat Equation: Alternating Direction Implicit (ADI) scheme.
Initial condition: multiple point heat sources.
"""
import numpy as np
import matplotlib.pyplot as plt
from scipy.linalg import solve_banded


# Domain and physical parameters
L, alpha = 1.0, 0.01
nx = ny = 61
dx = dy = L / (nx - 1)
dt = 0.005  # ADI is unconditionally stable, so dt is chosen for accuracy, not stability
r = alpha * dt / dx**2
nt = 100


x = np.linspace(0, L, nx)
y = np.linspace(0, L, ny)
u = np.zeros((ny, nx))
sources = [(0.25, 0.25), (0.75, 0.75), (0.25, 0.75), (0.75, 0.25)]  # multi-point sources
for sx, sy in sources:
    i, j = int(sy / dy), int(sx / dx)
    u[i, j] = 50.0


def thomas_solve(n, r):
    """Build and return tridiagonal coefficients for an implicit half-step."""
    ab = np.zeros((3, n))
    ab[0, 1:] = -r / 2
    ab[1, :] = 1 + r
    ab[2, :-1] = -r / 2
    return ab


ab_int = thomas_solve(nx - 2, r)


for n in range(nt):
    # Half-step 1: implicit in x, explicit in y
    u_star = u.copy()
    for i in range(1, ny - 1):
        rhs = (r / 2) * u[i, :-2] - (r - 1) * u[i, 1:-1] + (r / 2) * u[i, 2:]
        rhs += (r / 2) * (u[i - 1, 1:-1] - 2 * u[i, 1:-1] + u[i + 1, 1:-1])
        u_star[i, 1:-1] = solve_banded((1, 1), ab_int, rhs)
    # Half-step 2: implicit in y, explicit in x
    u_new = u_star.copy()
    for j in range(1, nx - 1):
        rhs = (r / 2) * u_star[:-2, j] - (r - 1) * u_star[1:-1, j] + (r / 2) * u_star[2:, j]
        u_new[1:-1, j] = solve_banded((1, 1), ab_int, rhs)
    u = u_new


X, Y = np.meshgrid(x, y)
plt.figure(figsize=(6, 5))
plt.contourf(X, Y, u, levels=30, cmap="hot")
plt.colorbar(label="Temperature")
plt.xlabel("x")
plt.ylabel("y")
plt.title("2D Heat Equation - ADI (Multi-Point Source IC)")
plt.tight_layout()
plt.savefig("heat_2d_adi_multisource.png", dpi=150)
plt.show()

"""
2D Heat Equation: Fourier spectral method (periodic BCs), exact in time via FFT.
Initial condition: sinusoidal wave pattern.
"""
import numpy as np
import matplotlib.pyplot as plt


# Domain and physical parameters
L, alpha = 2 * np.pi, 0.05
nx = ny = 128
x = np.linspace(0, L, nx, endpoint=False)
y = np.linspace(0, L, ny, endpoint=False)
X, Y = np.meshgrid(x, y)


u0 = np.sin(3 * X) * np.sin(2 * Y) + 0.5 * np.sin(5 * X)  # sinusoidal initial condition


kx = np.fft.fftfreq(nx, d=L / nx) * 2 * np.pi
ky = np.fft.fftfreq(ny, d=L / ny) * 2 * np.pi
KX, KY = np.meshgrid(kx, ky)
K2 = KX**2 + KY**2  # Laplacian eigenvalues in Fourier space


u0_hat = np.fft.fft2(u0)


t_final = 0.5  # No CFL restriction: solution advances via exact spectral decay factor
decay = np.exp(-alpha * K2 * t_final)
u_final = np.real(np.fft.ifft2(u0_hat * decay))


fig, axes = plt.subplots(1, 2, figsize=(11, 5))
im0 = axes[0].contourf(X, Y, u0, levels=30, cmap="coolwarm")
axes[0].set_title("Initial Condition (t=0)")
fig.colorbar(im0, ax=axes[0])
im1 = axes[1].contourf(X, Y, u_final, levels=30, cmap="coolwarm")
axes[1].set_title(f"Solution at t = {t_final}")
fig.colorbar(im1, ax=axes[1])
plt.tight_layout()
plt.savefig("heat_2d_fourier_spectral_sine.png", dpi=150)

Wave Equation Simulation Code Example 1
"""
1D Wave Equation: standard explicit FDTD leapfrog scheme, animated as a GIF.
Initial condition: Gaussian displacement, zero initial velocity.
"""
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation


# Domain and physical parameters
L, c = 1.0, 1.0
nx, nt = 201, 800
dx = L / (nx - 1)
dt = 0.5 * dx / c  # CFL condition: Courant number c*dt/dx <= 1
courant = (c * dt / dx) ** 2


x = np.linspace(0, L, nx)
u_prev = np.exp(-((x - L / 2) ** 2) / (2 * 0.03 ** 2))  # Gaussian initial displacement
v0 = np.zeros(nx)  # zero initial velocity


u_curr = u_prev.copy()
u_curr[1:-1] = (u_prev[1:-1] + dt * v0[1:-1]
                + 0.5 * courant * (u_prev[2:] - 2 * u_prev[1:-1] + u_prev[:-2]))
u_curr[0], u_curr[-1] = 0.0, 0.0


frames = [u_prev.copy(), u_curr.copy()]
for n in range(1, nt):
    u_next = np.zeros(nx)
    u_next[1:-1] = (2 * u_curr[1:-1] - u_prev[1:-1]
                    + courant * (u_curr[2:] - 2 * u_curr[1:-1] + u_curr[:-2]))  # leapfrog update
    u_next[0], u_next[-1] = 0.0, 0.0
    u_prev, u_curr = u_curr, u_next
    if n % 4 == 0:  # store every 4th step to keep the GIF short
        frames.append(u_curr.copy())


fig, ax = plt.subplots(figsize=(7, 5))
line, = ax.plot(x, frames[0])
ax.set_ylim(-1.1, 1.1)
ax.set_xlabel("x"); ax.set_ylabel("u(x,t)")
ax.set_title("1D Wave Equation - FDTD Leapfrog (Gaussian displacement)")


def update(frame):
    line.set_ydata(frame)
    return line,


ani = animation.FuncAnimation(fig, update, frames=frames, interval=30, blit=True)
ani.save("wave_1d_leapfrog_gaussian.gif", writer="pillow", fps=30)
plt.close(fig)

"""
1D Wave Equation: pseudo-spectral FFT method (periodic BCs), animated as a GIF.
Each Fourier mode is advanced by its exact analytic time solution.
Initial condition: zero displacement, localized initial velocity pulse.
"""
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation


# Domain and physical parameters
L, c = 2 * np.pi, 1.0
nx = 256
x = np.linspace(0, L, nx, endpoint=False)
dx = L / nx


u0 = np.zeros(nx)  # zero initial displacement
v0 = 5.0 * np.exp(-((x - L / 2) ** 2) / (2 * 0.1 ** 2))  # localized initial velocity pulse


k = np.fft.fftfreq(nx, d=dx) * 2 * np.pi  # wavenumber array
u0_hat = np.fft.fft(u0)
v0_hat = np.fft.fft(v0)
omega = c * k


times = np.linspace(0, 3.0, 150)  # spectral time-stepping is exact per mode, so no CFL restriction
frames = []
for t in times:
    with np.errstate(divide="ignore", invalid="ignore"):
        coeff = np.where(k != 0, v0_hat / np.where(omega == 0, 1, omega), 0.0)
    u_hat_t = u0_hat * np.cos(omega * t) + coeff * np.sin(omega * t)
    u_hat_t[0] = u0_hat[0] + v0_hat[0] * t  # k=0 mode grows linearly (mean velocity)
    frames.append(np.real(np.fft.ifft(u_hat_t)))


fig, ax = plt.subplots(figsize=(7, 5))
line, = ax.plot(x, frames[0])
ax.set_ylim(-0.2, 0.8)
ax.set_xlabel("x"); ax.set_ylabel("u(x,t)")
ax.set_title("1D Wave Equation - Pseudo-Spectral FFT (velocity pulse IC)")


def update(frame):
    line.set_ydata(frame)
    return line,


ani = animation.FuncAnimation(fig, update, frames=frames, interval=30, blit=True)
ani.save("wave_1d_spectral_velocity.gif", writer="pillow", fps=30)
plt.close(fig)

"""
2D Wave Equation: implicit Newmark-beta scheme (average-acceleration form),
unconditionally stable, animated as a GIF using a single LU factorization.
Initial condition: sinusoidal displacement plus a localized initial velocity pulse.
"""
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation
import scipy.sparse as sp
from scipy.sparse.linalg import splu


# Domain and physical parameters
L, c, nx, ny = 1.0, 1.0, 121, 121
dx = dy = L / (nx - 1)
x = np.linspace(0, L, nx)
y = np.linspace(0, L, ny)
X, Y = np.meshgrid(x, y, indexing="ij")
N = nx * ny


e = np.ones(nx)  # build sparse 2D Laplacian (Dirichlet BC) via Kronecker sum
D2x = sp.diags([e[:-1], -2 * e, e[:-1]], [-1, 0, 1]) / dx**2
Ix = sp.identity(nx)
Lap = sp.kron(D2x, Ix) + sp.kron(Ix, D2x)
K = -(c**2) * Lap  # positive semi-definite stiffness-like operator


u0 = 0.1 * np.sin(np.pi * X) * np.sin(np.pi * Y)  # sinusoidal initial displacement
sigma = 6 * dx  # pulse width kept well above grid spacing to avoid grid-scale aliasing
v0 = 3.0 * np.exp(-(((X - 0.3) ** 2 + (Y - 0.7) ** 2)) / (2 * sigma**2))
u = u0.flatten(); v = v0.flatten()
a = -K.dot(u)  # initial acceleration from the governing equation


beta, gamma, dt, nt = 0.3025, 0.6, 0.01, 150  # gamma>0.5 adds mild damping of grid-scale noise, stays unconditionally stable
A = sp.identity(N) + (dt**2) * beta * K
lu = splu(A.tocsc())  # factorize once since the system matrix is constant


frames = [u0.copy()]
for n in range(nt):
    rhs = u + dt * v + (dt**2) * (0.5 - beta) * a
    u_new = lu.solve(rhs)
    a_new = -K.dot(u_new)
    v_new = v + dt * ((1 - gamma) * a + gamma * a_new)
    u, v, a = u_new, v_new, a_new
    frames.append(u.reshape(nx, ny))

    im.set_array(frames[i].ravel())
    title.set_text(f"2D Wave Equation - Newmark-beta Implicit (t={i*dt:.2f}s)")
    return im, title


ani = animation.FuncAnimation(fig, update, frames=len(frames), interval=40, blit=False)

animated as a GIF. Initial condition: sinusoidal wave that steepens into a shock.

    fL, fR = 0.5 * uL**2, 0.5 * uR**2
    shock_speed = 0.5 * (uL + uR)

                     np.where(uL >= 0, fL, np.where(uR <= 0, fR, 0.0)),
                     np.where(shock_speed > 0, fL, fR))

"""

"""
2D Incompressible Navier-Stokes: Chorin's projection method on a collocated grid,
animated as a GIF. Lid-driven cavity flow, moderate Reynolds number.
"""
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation


# Domain and physical parameters
nx, ny, nu, rho = 71, 71, 0.05, 1.0
Lx, Ly = 1.0, 1.0
dx, dy = Lx / (nx - 1), Ly / (ny - 1)
x = np.linspace(0, Lx, nx)
y = np.linspace(0, Ly, ny)
X, Y = np.meshgrid(x, y, indexing="ij")


u = np.zeros((nx, ny)); v = np.zeros((nx, ny)); p = np.zeros((nx, ny))
u_lid = 1.0
dt = 0.2 * min(dx / u_lid, dx**2 / (4 * nu))  # advective + diffusive stability bound
nt, nit = 6200, 60


def pressure_poisson(p, b):
    """Jacobi-iterate the pressure Poisson equation with Neumann walls and one Dirichlet reference."""
    for _ in range(nit):
        pn = p.copy()
        p[1:-1, 1:-1] = ((pn[2:, 1:-1] + pn[:-2, 1:-1]) * dy**2
                          + (pn[1:-1, 2:] + pn[1:-1, :-2]) * dx**2
                          - b[1:-1, 1:-1] * dx**2 * dy**2) / (2 * (dx**2 + dy**2))
        p[:, -1] = p[:, -2]
        p[:, 0] = p[:, 1]
        p[0, :] = p[1, :]
        p[-1, :] = 0.0  # reference pressure to fix the null space
    return p

frames_uv = [(u.copy(), v.copy())]

    b = np.zeros((nx, ny))
    b[1:-1, 1:-1] = rho * (  # RHS built from the divergence of the tentative velocity
        (1 / dt) * ((un[2:, 1:-1] - un[:-2, 1:-1]) / (2 * dx) + (vn[1:-1, 2:] - vn[1:-1, :-2]) / (2 * dy))
        - ((un[2:, 1:-1] - un[:-2, 1:-1]) / (2 * dx)) ** 2
        - 2 * ((un[1:-1, 2:] - un[1:-1, :-2]) / (2 * dy) * (vn[2:, 1:-1] - vn[:-2, 1:-1]) / (2 * dx))
        - ((vn[1:-1, 2:] - vn[1:-1, :-2]) / (2 * dy)) ** 2
    )

        - dt / (2 * dx) * un[1:-1, 1:-1] * (un[2:, 1:-1] - un[:-2, 1:-1])

        - dt / (2 * rho * dy) * (p[1:-1, 2:] - p[1:-1, :-2])
        + nu * dt * ((vn[2:, 1:-1] - 2 * vn[1:-1, 1:-1] + vn[:-2, 1:-1]) / dx**2
                     + (vn[1:-1, 2:] - 2 * vn[1:-1, 1:-1] + vn[1:-1, :-2]) / dy**2))


    u[0, :] = 0; u[-1, :] = 0; u[:, 0] = 0; u[:, -1] = u_lid  # no-slip walls, moving lid

Navier Stokes Simulation Code Example 4 


"""
2D Navier-Stokes (vorticity-streamfunction form), near-inviscid high-Reynolds regime,
animated as a GIF. Pseudo-spectral FFT method with 2/3 dealiasing.
Initial condition: double shear layer (Kelvin-Helmholtz roll-up test case).
"""
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation


# Domain and physical parameters
N, L, nu = 256, 2 * np.pi, 1e-4  # near-inviscid: high Reynolds number regime
x = np.linspace(0, L, N, endpoint=False)
y = np.linspace(0, L, N, endpoint=False)
X, Y = np.meshgrid(x, y, indexing="ij")


delta, eps = 0.08, 0.05
u0 = np.where(Y <= np.pi, np.tanh((Y - np.pi / 2) / delta), np.tanh((3 * np.pi / 2 - Y) / delta))
v0 = eps * np.sin(X)  # small perturbation that seeds the Kelvin-Helmholtz roll-up


k = np.fft.fftfreq(N, d=L / N) * 2 * np.pi
KX, KY = np.meshgrid(k, k, indexing="ij")
K2 = KX**2 + KY**2
K2safe = np.where(K2 == 0, 1.0, K2)
kmax = N // 3
mask = (np.abs(KX) <= kmax) & (np.abs(KY) <= kmax)  # 2/3 dealiasing rule


def deriv(f_hat, K):
    return np.real(np.fft.ifft2(1j * K * f_hat))


omega_hat = np.fft.fft2(deriv(np.fft.fft2(v0), KX) - deriv(np.fft.fft2(u0), KY))


def velocity(omega_hat):
    psi_hat = omega_hat / K2safe
    psi_hat[0, 0] = 0.0
    return deriv(psi_hat, KY), -deriv(psi_hat, KX)


def rhs(omega_hat):
    u, v = velocity(omega_hat)
    omega_x = deriv(omega_hat, KX)
    omega_y = deriv(omega_hat, KY)
    nonlinear_hat = np.fft.fft2(u * omega_x + v * omega_y) * mask
    return -nonlinear_hat - nu * K2 * omega_hat


dx = L / N
nt = 400
frames = [np.real(np.fft.ifft2(omega_hat))]


for step in range(nt):
    u, v = velocity(omega_hat)
    speed_max = max(np.max(np.sqrt(u**2 + v**2)), 1e-6)
    dt = 0.3 * dx / speed_max  # CFL re-checked every step as the layer rolls up and speeds change
    k1 = rhs(omega_hat)
    k2 = rhs(omega_hat + dt * k1)
    omega_hat = omega_hat + dt / 2 * (k1 + k2)  # RK2 (Heun) time integration
    if step % 4 == 0:
        frames.append(np.real(np.fft.ifft2(omega_hat)))


fig, ax = plt.subplots(figsize=(6, 5))
im = ax.pcolormesh(X, Y, frames[0], cmap="RdBu_r", shading="gouraud", vmin=-12, vmax=12)
fig.colorbar(im, ax=ax, label="Vorticity")
ax.set_xlabel("x"); ax.set_ylabel("y")
ax.set_title("2D Navier-Stokes - Pseudo-Spectral Double Shear Layer")


def update(frame):
    im.set_array(frame.ravel())
    return im,


ani = animation.FuncAnimation(fig, update, frames=frames, interval=40, blit=True)
ani.save("navierstokes_2d_spectral_shearlayer.gif", writer="pillow", fps=25)
plt.close(fig)
